# Meson cross file TEMPLATE — iOS device, arm64 (Build 353, Libmpv pipeline).
#
# This file is a template, not a ready-to-use cross file: meson cross files
# cannot run commands, so the placeholders below are substituted by
# build_libmpv.py at runtime with the values `xcrun` reports on the actual
# CI runner, and the result is written to .cache353/cross/:
#
#   __SDK_PATH__   xcrun --sdk iphoneos --show-sdk-path
#   __SDK_NAME__   iphoneos
#   __CLANG__      xcrun --sdk iphoneos -f clang
#   __CLANGXX__    xcrun --sdk iphoneos -f clang++
#   __AR__         xcrun --sdk iphoneos -f ar
#   __RANLIB__     xcrun --sdk iphoneos -f ranlib
#   __STRIP__      xcrun --sdk iphoneos -f strip
#   __MIN_FLAG__   -mios-version-min=17.0
#
# Never hard-code /Applications/Xcode.app/... paths here; SDK locations
# differ between runners.

# objc is defined (same clang binary; objc_args below already carry
# -arch/-isysroot) because mpv's meson.build calls
# add_languages('objc', native: false) on every darwin configure once
# it detects the runner's macOS SDK (TOOLS/macos-sdk-version.py), and
# its audiounit/avfoundation sources are .m files.  objcpp is
# deliberately NOT defined: the only consumer in the chain was
# harfbuzz's gpu feature (util/meson.build calls
# add_languages('objcpp', required: true) on darwin), and that feature
# is disabled in the harfbuzz stage instead — it builds only the
# separate experimental libharfbuzz-gpu, which nothing here links.

[binaries]
c = '__CLANG__'
cpp = '__CLANGXX__'
objc = '__CLANG__'
ar = '__AR__'
ranlib = '__RANLIB__'
strip = '__STRIP__'
pkg-config = 'pkg-config'

[properties]
# No sys_root here on purpose: meson exports it as
# PKG_CONFIG_SYSROOT_DIR, and pkg-config then relocates EVERY absolute
# -I/-L path from .pc files into the SDK — including the vendored
# prefix's own paths (all dependencies live in <prefix>/lib/pkgconfig),
# which breaks e.g. harfbuzz's freetype2 lookup.  The SDK location
# reaches the compilers via -isysroot in the *_args below, and the one
# system library resolved through pkg-config (zlib) gets a hand-written
# .pc with absolute SDK paths from build_libmpv.py instead.
needs_exe_wrapper = true

[host_machine]
system = 'darwin'
cpu_family = 'aarch64'
cpu = 'arm64'
endian = 'little'

[built-in options]
c_args = ['-arch', 'arm64', '__MIN_FLAG__', '-isysroot', '__SDK_PATH__']
c_link_args = ['-arch', 'arm64', '__MIN_FLAG__', '-isysroot', '__SDK_PATH__']
cpp_args = ['-arch', 'arm64', '__MIN_FLAG__', '-isysroot', '__SDK_PATH__']
cpp_link_args = ['-arch', 'arm64', '__MIN_FLAG__', '-isysroot', '__SDK_PATH__']
objc_args = ['-arch', 'arm64', '__MIN_FLAG__', '-isysroot', '__SDK_PATH__']
objc_link_args = ['-arch', 'arm64', '__MIN_FLAG__', '-isysroot', '__SDK_PATH__']
